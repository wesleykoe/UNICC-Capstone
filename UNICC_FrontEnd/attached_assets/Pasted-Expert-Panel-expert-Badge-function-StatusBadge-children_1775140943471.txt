Badge 小组件
function StatusBadge({ children, tone = "neutral" }) {
  const styles = {
    neutral: "bg-slate-100 text-slate-700 border-slate-200",
    caution: "bg-amber-50 text-amber-700 border-amber-200",
    fail: "bg-rose-50 text-rose-700 border-rose-200",
    pass: "bg-emerald-50 text-emerald-700 border-emerald-200",
    info: "bg-blue-50 text-blue-700 border-blue-200",
  };

  return (
    <span className={`inline-flex items-center rounded-full border px-2.5 py-1 text-xs font-medium ${styles[tone]}`}>
      {children}
    </span>
  );
}
Expert Card
function ExpertCard({
  expertName,
  overallStatus,
  riskLevel,
  action,
  confidence,
  finding,
  frameworks = [],
  humanReviewRequired,
}) {
  const statusTone =
    overallStatus === "Pass"
      ? "pass"
      : overallStatus === "Fail"
      ? "fail"
      : "caution";

  const riskTone =
    riskLevel === "Low"
      ? "pass"
      : riskLevel === "Moderate"
      ? "info"
      : "caution";

  return (
    <div className="rounded-2xl border border-slate-200 bg-white p-6 shadow-sm">
      <div className="mb-4 flex items-start justify-between">
        <div>
          <h3 className="text-xl font-semibold text-slate-900">{expertName}</h3>
          <p className="mt-1 text-sm text-slate-500">
            Independent expert assessment
          </p>
        </div>
        <StatusBadge tone={statusTone}>{overallStatus}</StatusBadge>
      </div>

      <div className="mb-4 flex flex-wrap gap-2">
        <StatusBadge tone={riskTone}>Risk: {riskLevel}</StatusBadge>
        <StatusBadge tone="info">Action: {action}</StatusBadge>
        <StatusBadge tone="neutral">Confidence: {confidence}</StatusBadge>
      </div>

      <div className="mb-4 rounded-xl bg-slate-50 p-4">
        <div className="mb-1 text-xs font-semibold uppercase tracking-wide text-slate-500">
          Key Finding
        </div>
        <p className="text-sm leading-6 text-slate-700">{finding}</p>
      </div>

      <div className="mb-4">
        <div className="mb-2 text-xs font-semibold uppercase tracking-wide text-slate-500">
          Framework References
        </div>
        <div className="flex flex-wrap gap-2">
          {frameworks.length > 0 ? (
            frameworks.map((fw) => (
              <span
                key={fw}
                className="rounded-full bg-slate-100 px-3 py-1 text-xs text-slate-700"
              >
                {fw}
              </span>
            ))
          ) : (
            <span className="text-sm text-slate-500">
              Standard expert framework references applied
            </span>
          )}
        </div>
      </div>

      <div className="border-t border-slate-200 pt-4">
        <div className="flex items-center justify-between text-sm">
          <span className="text-slate-500">Human Review</span>
          <span
            className={`font-medium ${
              humanReviewRequired ? "text-amber-700" : "text-emerald-700"
            }`}
          >
            {humanReviewRequired ? "Required" : "Not Required"}
          </span>
        </div>
      </div>
    </div>
  );
}
