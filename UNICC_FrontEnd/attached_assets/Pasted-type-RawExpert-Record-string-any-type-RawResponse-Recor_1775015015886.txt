type RawExpert = Record<string, any>;
type RawResponse = Record<string, any>;

function normalizeExpertName(name: string, key: string) {
  const source = `${name || ""} ${key || ""}`.toLowerCase();

  if (source.includes("governance")) return "Governance Expert";
  if (source.includes("threat")) return "Threat Expert";
  if (source.includes("behavior")) return "Behavioral Expert";

  if (key.toLowerCase().includes("governance")) return "Governance Expert";
  if (key.toLowerCase().includes("threat")) return "Threat Expert";
  if (key.toLowerCase().includes("behavioral")) return "Behavioral Expert";

  return key || "Unknown Expert";
}

function toArray(value: any): string[] {
  if (!value) return [];
  if (Array.isArray(value)) return value.filter(Boolean).map(String);
  return [String(value)];
}

function normalizeExpert(key: string, expert: RawExpert) {
  const mergedFrameworks = [
    ...toArray(expert.framework_references),
    ...toArray(expert.framework_reference),
  ];

  return {
    expert_name: normalizeExpertName(expert.expert_name, key),
    overall_status: expert.overall_status || "Caution",
    risk_level: expert.risk_level || "Moderate",
    recommended_action: expert.recommended_action || "Escalate",
    requires_human_review: Boolean(expert.requires_human_review),
    confidence_level: expert.confidence_level || "Moderate",
    rationale_summary: expert.rationale_summary || "No rationale provided.",
    framework_references: [...new Set(mergedFrameworks)],
  };
}

export function normalizeResponse(raw: RawResponse) {
  const rawExperts = raw?.expert_outputs || {};

  const experts = Object.entries(rawExperts).map(([key, value]) =>
    normalizeExpert(key, value as RawExpert)
  );

  const finalRec = raw?.final_council_recommendation || {};
  const metadata = raw?.council_metadata || {};

  return {
    council_metadata: {
      council_run_id: metadata.council_run_id || "N/A",
      timestamp: metadata.timestamp || "",
      evaluation_method: metadata.evaluation_method || "council",
      evaluation_time_seconds: metadata.evaluation_time_seconds || 0,
      slm_version: metadata.slm_version || "Unknown",
      fine_tune_version: metadata.fine_tune_version || "Unknown",
      arbitration_rule_set_version:
        metadata.arbitration_rule_set_version || "Unknown",
      experts_invoked: metadata.experts_invoked || [],
      risks_detected: metadata.risks_detected || 0,
      high_severity_risks: metadata.high_severity_risks || 0,
      final_recommendation: metadata.final_recommendation || "revise",
      developer_confidence_score: metadata.developer_confidence_score || 3,
      input_hash: metadata.input_hash || "",
      notes: metadata.notes || "",
    },
    expert_outputs: experts,
    final_council_recommendation: {
      final_decision: finalRec.final_decision || "Revise",
      final_risk_level: finalRec.final_risk_level || "Moderate",
      consensus_level: finalRec.consensus_level || "Majority Agreement",
      summary_of_key_disagreements:
        finalRec.summary_of_key_disagreements || [],
      dominant_expert_influence:
        finalRec.dominant_expert_influence || "Mixed",
      human_review_required: Boolean(finalRec.human_review_required),
      conditions_for_approval: finalRec.conditions_for_approval || [],
      mitigation_requirements: finalRec.mitigation_requirements || [],
      confidence_level: finalRec.confidence_level || "Moderate",
      final_rationale: finalRec.final_rationale || "No rationale provided.",
      cited_frameworks: finalRec.cited_frameworks || [],
    },
  };
}

export function getDecisionTone(decision: string) {
  const d = decision.toLowerCase();
  if (d === "approve") return "success";
  if (d === "revise") return "warning";
  if (d === "escalate") return "warning";
  if (d === "reject") return "danger";
  return "neutral";
}

export function getStatusTone(status: string) {
  const s = status.toLowerCase();
  if (s === "pass") return "success";
  if (s === "caution") return "warning";
  if (s === "fail") return "danger";
  return "neutral";
}
