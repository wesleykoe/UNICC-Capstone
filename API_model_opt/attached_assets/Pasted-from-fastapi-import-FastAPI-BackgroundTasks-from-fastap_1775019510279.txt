from fastapi import FastAPI, BackgroundTasks
from fastapi.middleware.cors import CORSMiddleware
import uuid
import time

app = FastAPI()

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

        jobs[job_id]["status"] = "running"

        jobs[job_id]["current_step"] = "governance"

        jobs[job_id]["threat"] = threat_result

        jobs[job_id]["current_step"] = "behavioral"
        behavioral_result = run_behavioral_expert(payload)  # 你自己的函数
        jobs[job_id]["behavioral"] = behavioral_result

        # 4. Final arbitration
        jobs[job_id]["current_step"] = "finalizing"

        jobs[job_id]["final"] = final_result

@app.post("/evaluate")
async def evaluate(payload: dict, background_tasks: BackgroundTasks):
    job_id = str(uuid.uuid4())

    jobs[job_id] = {
        "status": "queued",
        "current_step": "starting",
        "governance": None,
        "threat": None,
        "behavioral": None,
        "final": None,
        "error": None,
    }

    background_tasks.add_task(run_full_evaluation, job_id, payload)

    return {
        "job_id": job_id,
        "status": "queued"
    }

@app.get("/status/{job_id}")
async def get_status(job_id: str):
    return jobs.get(job_id, {"status": "not_found"})
