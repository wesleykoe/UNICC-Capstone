import React from "react";
import {
  AlertTriangle,
  CheckCircle2,
  ShieldAlert,
  ClipboardList,
  Users,
  Scale,
  ArrowRightCircle,
  FileWarning,
  Info,
} from "lucide-react";

type ExpertOutput = {
  expert_name?: string;
  overall_status?: "Pass" | "Caution" | "Fail" | string;
  risk_level?: "Low" | "Moderate" | "High" | "Critical" | string;
  recommended_action?: "Approve" | "Revise" | "Escalate" | "Reject" | string;
  requires_human_review?: boolean;
  confidence_level?: "Low" | "Moderate" | "High" | string;
  rationale_summary?: string;
  framework_references?: string[];
};

type FinalCouncilRecommendation = {
  final_decision?: "Approve" | "Revise" | "Escalate" | "Reject" | string;
  final_risk_level?: "Low" | "Moderate" | "High" | "Critical" | string;
  consensus_level?: "Full Agreement" | "Majority Agreement" | "Structured Disagreement" | string;
  summary_of_key_disagreements?: string[];
  dominant_expert_influence?: "Governance" | "Threat" | "Behavioral" | "Mixed" | string;
  human_review_required?: boolean;
  conditions_for_approval?: string[];
  mitigation_requirements?: string[];
  confidence_level?: "Low" | "Moderate" | "High" | string;
  final_rationale?: string;
  cited_frameworks?: string[];
};

type CouncilMetadata = {
  council_run_id?: string;
  timestamp?: string;
  evaluation_time_seconds?: number;
  slm_version?: string;
  fine_tune_version?: string;
  arbitration_rule_set_version?: string;
  final_recommendation?: string;
};

type Props = {
  result: {
    council_metadata?: CouncilMetadata;
    expert_outputs?: Record<string, ExpertOutput>;
    final_council_recommendation?: FinalCouncilRecommendation;
  };
};

    badge: string;
    iconBg: string;
    iconColor: string;
    title: string;
    subtitle: string;
    icon: React.ReactNode;
  }
> = {
  Approve: {
    container: "border-emerald-200 bg-emerald-50/70",
    badge: "bg-emerald-100 text-emerald-800 border border-emerald-200",
    iconBg: "bg-emerald-100",
    iconColor: "text-emerald-700",
    title: "Deployment Approved",
    subtitle: "Deployment may proceed with standard monitoring.",
    icon: <CheckCircle2 className="h-6 w-6" />,
  },
  Revise: {
    container: "border-amber-200 bg-amber-50/70",
    badge: "bg-amber-100 text-amber-800 border border-amber-200",
    iconBg: "bg-amber-100",
    iconColor: "text-amber-700",
    title: "Revision Required",
    subtitle: "Deployment requires targeted revision before approval.",
    icon: <ClipboardList className="h-6 w-6" />,
  },
  Escalate: {
    container: "border-orange-200 bg-orange-50/70",
    badge: "bg-orange-100 text-orange-800 border border-orange-200",
    iconBg: "bg-orange-100",
    iconColor: "text-orange-700",
    title: "Escalation Required",
    subtitle: "Deployment should not proceed without senior human review.",
    icon: <ShieldAlert className="h-6 w-6" />,
  },
  Reject: {
    container: "border-red-200 bg-red-50/70",
    badge: "bg-red-100 text-red-800 border border-red-200",
    iconBg: "bg-red-100",
    iconColor: "text-red-700",
    title: "Deployment Not Approved",
    subtitle: "Deployment should not proceed in its current form.",
    icon: <AlertTriangle className="h-6 w-6" />,
  },
};

const RISK_THEME: Record<string, string> = {
  Low: "bg-emerald-50 text-emerald-700 border border-emerald-200",
  Moderate: "bg-amber-50 text-amber-700 border border-amber-200",
  High: "bg-orange-50 text-orange-700 border border-orange-200",
  Critical: "bg-red-50 text-red-700 border border-red-200",
};

const STATUS_THEME: Record<string, string> = {
  Pass: "bg-emerald-50 text-emerald-700 border border-emerald-200",
  Caution: "bg-amber-50 text-amber-700 border border-amber-200",
  Fail: "bg-red-50 text-red-700 border border-red-200",
};

const ACTION_THEME: Record<string, string> = {
  Approve: "bg-emerald-50 text-emerald-700 border border-emerald-200",
  Revise: "bg-amber-50 text-amber-700 border border-amber-200",
  Escalate: "bg-orange-50 text-orange-700 border border-orange-200",
  Reject: "bg-red-50 text-red-700 border border-red-200",
};

function formatDecisionLabel(value?: string) {
  if (!value) return "Unknown";
  return value;
}

function formatDate(iso?: string) {
  if (!iso) return "—";
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return iso;
  return d.toLocaleString();
}

function sentenceCase(value?: string) {
  if (!value) return "—";
  return value;
}

function getDecisionTheme(decision?: string) {
  return DECISION_THEME[decision || ""] || {
    container: "border-slate-200 bg-slate-50",
    badge: "bg-slate-100 text-slate-800 border border-slate-200",
    iconBg: "bg-slate-100",
    iconColor: "text-slate-700",
    title: "Council Decision",
    subtitle: "Decision available.",
    icon: <Info className="h-6 w-6" />,
  };
}

function getExpertPlainMeaning(expertName?: string, expert?: ExpertOutput) {
  const status = expert?.overall_status;
  const risk = expert?.risk_level;

  if (expertName?.toLowerCase().includes("governance")) {
    if (status === "Fail") return "Institutional or neutrality concerns make this unsafe for deployment.";
    if (status === "Caution") return "Governance alignment is not yet strong enough for direct deployment.";
    return "Governance review did not identify blocking institutional concerns.";
  }

  if (expertName?.toLowerCase().includes("threat")) {
    if (status === "Fail") return "Security or adversarial weaknesses create a meaningful exploitation risk.";
    if (status === "Caution") return "Some misuse or robustness concerns still require mitigation.";
    return "No major exploitability concerns were identified in this review.";
  }

  if (expertName?.toLowerCase().includes("behavioral")) {
    if (status === "Fail") return "Behavior under stress or variation is not reliable enough for deployment.";
    if (status === "Caution") return "Behavior remains partly unstable and needs further tuning.";
    return "Behavior appears sufficiently stable and aligned in the evaluated context.";
  }

  if (risk === "Critical" || risk === "High") {
    return "This expert identified a high-priority concern affecting deployment readiness.";
  }
  if (status === "Pass") {
    return "This expert did not identify a blocking issue.";
  }
  return "This expert recommends caution before deployment.";
}

function getArbitrationExplanation(
  experts: Record<string, ExpertOutput> | undefined,
  finalDecision?: string
) {
  if (!experts) return "Council decision generated from combined expert review.";

  const entries = Object.entries(experts);
  const actions = entries.map(([, v]) => v?.recommended_action);
  const statuses = entries.map(([, v]) => v?.overall_status);

  if (actions.includes("Reject")) {
    const who = entries.find(([, v]) => v?.recommended_action === "Reject")?.[0];
    return `${who || "One expert"} issued Reject, so the final council decision is Reject.`;
  }
  if (actions.includes("Escalate")) {
    const who = entries.find(([, v]) => v?.recommended_action === "Escalate")?.[0];
    return `${who || "One expert"} requested Escalate, so the case requires senior review.`;
  }
  if (statuses.includes("Fail")) {
    const who = entries.find(([, v]) => v?.overall_status === "Fail")?.[0];
    return `${who || "One expert"} returned Fail, which triggered escalation.`;
  }
  if (actions.includes("Revise") || statuses.includes("Caution")) {
    return "Caution or revision findings were identified, so deployment requires changes before approval.";
  }
  if (finalDecision === "Approve") {
    return "All expert reviews cleared the deployment path, so the system is approved.";
  }
  return "Council decision generated from deterministic arbitration across the expert panel.";
}

function Pill({
  label,
  value,
}: {
  label: string;
  value: React.ReactNode;
}) {
  return (
    <div className="rounded-2xl border border-slate-200 bg-white px-4 py-3 shadow-sm">
      <div className="text-xs font-medium uppercase tracking-wide text-slate-500">
        {label}
      </div>
      <div className="mt-1 text-sm font-semibold text-slate-900">{value}</div>
    </div>
  );
}

function Badge({
  children,
  className = "",
}: {
  children: React.ReactNode;
  className?: string;
}) {
  return (
    <span
      className={`inline-flex items-center rounded-full px-2.5 py-1 text-xs font-semibold ${className}`}
    >
      {children}
    </span>
  );
}

function SectionCard({
  title,
  icon,
  children,
}: {
  title: string;
  icon: React.ReactNode;
  children: React.ReactNode;
}) {
  return (
    <div className="rounded-3xl border border-slate-200 bg-white p-6 shadow-sm">
      <div className="mb-4 flex items-center gap-3">
        <div className="rounded-xl bg-slate-100 p-2 text-slate-700">{icon}</div>
        <h3 className="text-lg font-semibold text-slate-900">{title}</h3>
      </div>
      {children}
    </div>
  );
}

export default function CouncilDecisionPanel({ result }: Props) {
  const metadata = result?.council_metadata || {};
  const experts = result?.expert_outputs || {};
  const finalRec = result?.final_council_recommendation || {};

  const finalDecision = formatDecisionLabel(finalRec.final_decision);
  const theme = getDecisionTheme(finalDecision);

  const whyText =
    finalRec.final_rationale ||
    "The council identified material concerns that affect deployment readiness.";

  const operationalImpact =
    finalDecision === "Reject"
      ? "This system should not be deployed in live humanitarian workflows until the flagged issues are redesigned and re-evaluated."
      : finalDecision === "Escalate"
      ? "This case requires senior human review before any operational deployment decision is made."
      : finalDecision === "Revise"
      ? "The system may become deployable after the listed issues are addressed and resubmitted for council review."
      : "The system may proceed to deployment with standard monitoring and governance oversight.";

  const nextSteps = [
    ...(finalRec.conditions_for_approval || []),
    ...(finalRec.mitigation_requirements || []),
  ];

  const arbitrationExplanation = getArbitrationExplanation(experts, finalDecision);

  return (
    <div className="space-y-6">
      {/* HERO */}
      <section className={`rounded-3xl border p-6 shadow-sm ${theme.container}`}>
        <div className="flex flex-col gap-6 lg:flex-row lg:items-start lg:justify-between">
          <div className="flex items-start gap-4">
            <div className={`rounded-2xl p-3 ${theme.iconBg} ${theme.iconColor}`}>
              {theme.icon}
            </div>

            <div>
              <div className="mb-3">
                <Badge className={theme.badge}>{finalDecision}</Badge>
              </div>
              <h1 className="text-3xl font-bold tracking-tight text-slate-950">
                {theme.title}
              </h1>
              <p className="mt-2 max-w-2xl text-sm leading-6 text-slate-700">
                {theme.subtitle}
              </p>
            </div>
          </div>

          <div className="grid w-full grid-cols-2 gap-3 lg:w-auto lg:min-w-[340px]">
            <Pill label="Risk Level" value={finalRec.final_risk_level || "—"} />
            <Pill label="Consensus" value={finalRec.consensus_level || "—"} />
            <Pill
              label="Primary Driver"
              value={finalRec.dominant_expert_influence || "—"}
            />
            <Pill
              label="Human Review"
              value={finalRec.human_review_required ? "Required" : "Not Required"}
            />
          </div>
        </div>
      </section>

      {/* 3 PART SUMMARY */}
      <section className="grid grid-cols-1 gap-6 xl:grid-cols-3">
        <SectionCard title="Why this decision" icon={<Scale className="h-5 w-5" />}>
          <p className="text-sm leading-7 text-slate-700">{whyText}</p>
        </SectionCard>

        <SectionCard
          title="Operational impact"
          icon={<FileWarning className="h-5 w-5" />}
        >
          <p className="text-sm leading-7 text-slate-700">{operationalImpact}</p>
        </SectionCard>

        <SectionCard
          title="Required next steps"
          icon={<ArrowRightCircle className="h-5 w-5" />}
        >
          {nextSteps.length > 0 ? (
            <ul className="space-y-3 text-sm leading-6 text-slate-700">
              {nextSteps.map((item, idx) => (
                <li key={`${item}-${idx}`} className="flex gap-3">
                  <span className="mt-1 h-2 w-2 rounded-full bg-slate-400" />
                  <span>{item}</span>
                </li>
              ))}
            </ul>
          ) : (
            <p className="text-sm leading-7 text-slate-700">
              No additional remediation steps were returned for this decision.
            </p>
          )}
        </SectionCard>
      </section>

      {/* EXPLAINABILITY */}
      <SectionCard
        title="How the council reached this decision"
        icon={<Users className="h-5 w-5" />}
      >
        <div className="mb-5 rounded-2xl border border-slate-200 bg-slate-50 px-4 py-3">
          <div className="text-sm font-semibold text-slate-900">Arbitration logic</div>
          <p className="mt-1 text-sm leading-6 text-slate-700">
            {arbitrationExplanation}
          </p>
        </div>

        <div className="grid grid-cols-1 gap-4 xl:grid-cols-3">
          {Object.entries(experts).map(([expertKey, expert]) => {
            const statusClass = STATUS_THEME[expert.overall_status || ""] || "bg-slate-50 text-slate-700 border border-slate-200";
            const riskClass = RISK_THEME[expert.risk_level || ""] || "bg-slate-50 text-slate-700 border border-slate-200";
            const actionClass = ACTION_THEME[expert.recommended_action || ""] || "bg-slate-50 text-slate-700 border border-slate-200";

            return (
              <div
                key={expertKey}
                className="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm"
              >
                <div className="flex items-start justify-between gap-3">
                  <div>
                    <h4 className="text-base font-semibold text-slate-900">
                      {expert.expert_name || expertKey}
                    </h4>
                    <p className="mt-1 text-sm leading-6 text-slate-600">
                      {getExpertPlainMeaning(expertKey, expert)}
                    </p>
                  </div>
                </div>

                <div className="mt-4 flex flex-wrap gap-2">
                  <Badge className={statusClass}>
                    Status: {expert.overall_status || "—"}
                  </Badge>
                  <Badge className={riskClass}>
                    Risk: {expert.risk_level || "—"}
                  </Badge>
                  <Badge className={actionClass}>
                    Action: {expert.recommended_action || "—"}
                  </Badge>
                </div>

                <div className="mt-4 rounded-2xl bg-slate-50 p-4">
                  <div className="text-xs font-semibold uppercase tracking-wide text-slate-500">
                    Rationale summary
                  </div>
                  <p className="mt-2 text-sm leading-6 text-slate-700">
                    {sentenceCase(expert.rationale_summary)}
                  </p>
                </div>

                {!!expert.framework_references?.length && (
                  <div className="mt-4">
                    <div className="mb-2 text-xs font-semibold uppercase tracking-wide text-slate-500">
                      Referenced frameworks
                    </div>
                    <div className="flex flex-wrap gap-2">
                      {expert.framework_references.map((fw, idx) => (
                        <span
                          key={`${fw}-${idx}`}
                          className="rounded-full border border-slate-200 bg-slate-50 px-2.5 py-1 text-xs text-slate-700"
                        >
                          {fw}
                        </span>
                      ))}
                    </div>
                  </div>
                )}
              </div>
            );
          })}
        </div>
      </SectionCard>

      {/* DISAGREEMENTS */}
      {!!finalRec.summary_of_key_disagreements?.length && (
        <SectionCard
          title="Key disagreements across experts"
          icon={<AlertTriangle className="h-5 w-5" />}
        >
          <ul className="space-y-3 text-sm leading-6 text-slate-700">
            {finalRec.summary_of_key_disagreements.map((item, idx) => (
              <li key={`${item}-${idx}`} className="flex gap-3">
                <span className="mt-1 h-2 w-2 rounded-full bg-slate-400" />
                <span>{item}</span>
              </li>
            ))}
          </ul>
        </SectionCard>
      )}

      {/* FOOTER META */}
      <section className="rounded-3xl border border-slate-200 bg-white p-6 shadow-sm">
        <h3 className="text-lg font-semibold text-slate-900">Run metadata</h3>

        <div className="mt-4 grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
          <Pill label="Run ID" value={metadata.council_run_id || "—"} />
          <Pill label="Timestamp" value={formatDate(metadata.timestamp)} />
          <Pill
            label="Evaluation Time"
            value={
              metadata.evaluation_time_seconds !== undefined
                ? `${metadata.evaluation_time_seconds}s`
                : "—"
            }
          />
          <Pill label="Model" value={metadata.slm_version || "—"} />
          <Pill label="Fine-Tune Version" value={metadata.fine_tune_version || "—"} />
          <Pill
            label="Arbitration Version"
            value={metadata.arbitration_rule_set_version || "—"}
          />
        </div>
      </section>
    </div>
  );
}
