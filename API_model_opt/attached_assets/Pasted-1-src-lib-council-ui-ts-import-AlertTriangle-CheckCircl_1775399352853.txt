import {

import type { LucideIcon } from "lucide-react";

export type ExpertAction = "Approve" | "Revise" | "Escalate" | "Reject";
export type ExpertStatus = "Pass" | "Caution" | "Fail";
export type RiskLevel = "Low" | "Moderate" | "High" | "Critical";
export type ConfidenceLevel = "Low" | "Moderate" | "High";

export function getRiskClasses(risk: string) {
  switch (risk) {
    case "Low":
      return "bg-emerald-50 text-emerald-700 border border-emerald-200";
    case "Moderate":
      return "bg-amber-50 text-amber-700 border border-amber-200";
    case "High":
      return "bg-red-50 text-red-700 border border-red-200";
    case "Critical":
      return "bg-rose-100 text-rose-800 border border-rose-300";
    default:
      return "bg-slate-100 text-slate-700 border border-slate-200";
  }
}

export function getActionClasses(action: string) {
  switch (action) {
    case "Approve":
      return "bg-emerald-50 text-emerald-700 border border-emerald-200";
    case "Revise":
      return "bg-amber-50 text-amber-700 border border-amber-200";
    case "Escalate":
      return "bg-orange-50 text-orange-700 border border-orange-200";
    case "Reject":
      return "bg-red-50 text-red-700 border border-red-200";
    default:
      return "bg-slate-100 text-slate-700 border border-slate-200";
  }
}

export function getStatusClasses(status: string) {
  switch (status) {
    case "Pass":
      return "bg-emerald-50 text-emerald-700 border border-emerald-200";
    case "Caution":
      return "bg-amber-50 text-amber-700 border border-amber-200";
    case "Fail":
      return "bg-red-50 text-red-700 border border-red-200";
    default:
      return "bg-slate-100 text-slate-700 border border-slate-200";

}

export function getDecisionIcon(decision: string): LucideIcon {
  switch (decision) {
    case "Approve":
      return CheckCircle2;
    case "Revise":
      return FileWarning;
    case "Escalate":
      return ShieldAlert;
    case "Reject":
      return XCircle;
    default:
      return CircleHelp;
  }
}

export function getDecisionHeroClasses(decision: string) {
  switch (decision) {
    case "Approve":
      return "bg-emerald-50 border-emerald-200 text-emerald-900";
    case "Revise":
      return "bg-amber-50 border-amber-200 text-amber-900";

      return "bg-orange-50 border-orange-200 text-orange-900";
    case "Reject":
      return "bg-red-50 border-red-200 text-red-900";
    default:
      return "bg-slate-50 border-slate-200 text-slate-900";
  }
}

export function getDecisionMeaning(decision: string) {
  switch (decision) {
    case "Approve":
      return "Deployment is acceptable under the current review conditions.";
    case "Revise":
      return "Deployment is not yet ready and requires targeted revisions.";
    case "Escalate":
      return "Deployment requires senior review before any operational use.";
    case "Reject":
      return "Deployment is not permitted at this time.";
    default:
      return "Decision pending additional review.";
  }
}

export function getHumanInterpretation(
  expertName: string,
  status: string,
  risk: string,
  action: string
) {
  if (expertName.includes("Governance")) {
    if (action === "Approve") return "System appears institutionally acceptable for deployment.";
    if (action === "Revise") return "System is broadly acceptable but needs governance safeguards.";
    if (action === "Escalate") return "Institutional legitimacy concerns require senior oversight.";
    return "System is not acceptable within current governance boundaries.";
  }

  if (expertName.includes("Threat")) {
    if (action === "Approve") return "No major exploitability concerns were detected.";
    if (action === "Revise") return "Some security controls should be strengthened before deployment.";
    if (action === "Escalate") return "Threat exposure requires specialist review.";
    return "Security or misuse risk is too high for deployment.";
  }

  if (expertName.includes("Behavioral")) {
    if (action === "Approve") return "Model behavior appears stable and aligned across tested contexts.";
    if (action === "Revise") return "Behavior is mostly acceptable but needs tighter alignment controls.";
    if (action === "Escalate") return "Unsafe behavioral drift requires senior review.";
    return "Model behavior is unsafe under the evaluated conditions.";
  }

  if (status === "Pass" && risk === "Low") return "Assessment outcome is favorable.";
  if (action === "Reject") return "Assessment outcome indicates unacceptable risk.";
  return "Assessment requires additional attention.";
}

export function getAppliedRule(actions: string[], statuses: string[]) {
  if (actions.includes("Reject")) return "Applied rule: Any Reject → Final Reject";
  if (actions.includes("Escalate") || statuses.includes("Fail")) {
    return "Applied rule: Any Escalate or Fail → Final Escalate";
  }
  if (actions.includes("Revise") || statuses.includes("Caution")) {
    return "Applied rule: Any Revise or Caution → Final Revise";
  }
  return "Applied rule: All Pass → Final Approve";
}

export function summarizeImpact(finalRisk: string, dominantExpert: string, decision: string) {
  if (decision === "Reject") {
    return [
      "System may create operational, ethical, or reputational harm if deployed now.",
      "Current safeguards are insufficient for a humanitarian or UN-style context.",
    ];
  }
  if (decision === "Escalate") {
    return [
      "System requires senior review before operational release.",
      "Current assessment indicates risks that should not be handled at the team level alone.",
    ];
  }
  if (decision === "Revise") {
    return [
      "System is promising but not yet ready for deployment.",
      "Targeted mitigation is needed before a clean approval can be issued.",
    ];
  }
  return [
    "System is acceptable for the current deployment context.",
    "Standard monitoring and governance controls should still remain in place.",
  ];
}

export function summarizeWhy(expertOutputs: {
  expert_name?: string;
  expertName?: string;
  rationale_summary?: string;
  recommended_action?: string;
  risk_level?: string;
}[]) {
  return expertOutputs
    .map((e) => ({
      title: e.expert_name || e.expertName || "Expert",
      text: e.rationale_summary || "No rationale provided.",
      action: e.recommended_action || "N/A",
      risk: e.risk_level || "N/A",
    }))
    .slice(0, 3);
}
