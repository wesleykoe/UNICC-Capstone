先引入图标
import {
  CheckCircle2,
  TriangleAlert,
  PencilLine,
  ShieldAlert,
} from "lucide-react";
2）写一个 decision 配置函数
function getDecisionConfig(decision?: string) {
  switch (decision) {
    case "Approve":
      return {
        icon: CheckCircle2,
        iconClass: "text-emerald-600",
        badgeClass: "border-emerald-200 bg-emerald-50 text-emerald-700",
        panelClass: "border-emerald-200 bg-emerald-50",
      };
    case "Revise":
      return {
        icon: PencilLine,
        iconClass: "text-blue-600",
        badgeClass: "border-blue-200 bg-blue-50 text-blue-700",
        panelClass: "border-blue-200 bg-blue-50",
      };
    case "Escalate":
      return {
        icon: ShieldAlert,
        iconClass: "text-amber-600",
        badgeClass: "border-amber-200 bg-amber-50 text-amber-700",
        panelClass: "border-amber-200 bg-amber-50",
      };
    case "Reject":
    default:
      return {
        icon: TriangleAlert,
        iconClass: "text-red-600",
        badgeClass: "border-red-200 bg-red-50 text-red-700",
        panelClass: "border-red-200 bg-red-50",
      };
  }
}
3）在 Council hero 里用它
const decisionValue = currentCouncilView?.final_decision || "Reject";
const decisionConfig = getDecisionConfig(decisionValue);
const DecisionIcon = decisionConfig.icon;
