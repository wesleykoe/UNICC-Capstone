1）把主容器改宽，不要再缩在中间

如果你现在有类似：

<div className="mx-auto max-w-5xl px-6">

让它改成：

<div className="mx-auto w-full max-w-[1360px] px-8 xl:px-10">

如果是更大的页面，比如 Dashboard / Council，可以直接：

<div className="mx-auto w-full max-w-[1440px] px-8 xl:px-12">
你可以直接给它这段要求：
Please replace narrow page containers like max-w-4xl / max-w-5xl / max-w-6xl
with a wider responsive layout such as:
- max-w-[1360px] for standard pages
- max-w-[1440px] for dashboard and council views
Also increase horizontal padding so the page feels fuller and more premium.
2）Council 页面结构，直接这样改

让它把 Council Decision tab 改成这种结构：

function CouncilDecisionTab({ selectedRun }) {
  const [selectedExperts, setSelectedExperts] = useState([
    "Governance Expert",
    "Threat Expert",
    "Behavioral Expert",
  ]);

  const expertOutputs = selectedRun?.data?.expert_outputs || {};
  const decision = selectedRun?.data?.final_council_recommendation || {};
  const visibleExperts = Object.entries(expertOutputs).filter(([name]) =>
    selectedExperts.includes(name)
  );

  function toggleExpert(name) {
    setSelectedExperts((prev) =>
      prev.includes(name)
        ? prev.filter((x) => x !== name)
        : [...prev, name]
    );
  }

  return (
    <div className="mx-auto w-full max-w-[1440px] px-8 xl:px-12 py-8 space-y-8">
      <div>
        <h1 className="text-4xl font-semibold tracking-tight">Council Decision</h1>
        <p className="mt-2 text-base text-slate-500">
          Final recommendation from the UNICC AI Safety Council
        </p>
      </div>

      {/* expert selector row */}
      <div className="rounded-2xl border bg-white p-5">
        <div className="mb-3 text-sm font-medium text-slate-600">
          Select experts for council view
        </div>
        <div className="flex flex-wrap gap-3">
          {["Governance Expert", "Threat Expert", "Behavioral Expert"].map((expert) => {
            const active = selectedExperts.includes(expert);
            return (
              <button
                key={expert}
                onClick={() => toggleExpert(expert)}
                className={`rounded-xl border px-4 py-3 text-sm font-medium transition ${
                  active
                    ? "border-blue-500 bg-blue-50 text-blue-700"
                    : "border-slate-200 bg-white text-slate-600 hover:border-slate-300"
                }`}
              >
                {expert}
              </button>
            );
          })}
        </div>
      </div>

      {/* hero decision */}
      <div className="rounded-3xl border bg-white p-8 shadow-sm">
        <div className="rounded-2xl border border-red-200 bg-red-50 px-8 py-10 text-center">
          <div className="text-sm font-semibold uppercase tracking-[0.18em] text-slate-500">
            Final Decision
          </div>
          <div className="mt-3 text-6xl font-semibold text-red-600">
            {decision.final_decision || "Reject"}
          </div>
          <div className="mt-6 flex flex-wrap items-center justify-center gap-3">
            <span className="rounded-full border bg-white px-4 py-2 text-sm">
              Risk {decision.final_risk_level || "High"}
            </span>
            <span className="rounded-full border bg-white px-4 py-2 text-sm">
              Consensus {decision.consensus_level || "Majority Agreement"}
            </span>
            <span className="rounded-full border bg-white px-4 py-2 text-sm">
              Experts Reviewed {visibleExperts.length}/3
            </span>
            <span className="rounded-full border bg-amber-50 px-4 py-2 text-sm text-amber-700">
              Human Review {decision.human_review_required ? "Required" : "Not Required"}
            </span>
          </div>
        </div>

        <div className="mt-8 grid gap-6 lg:grid-cols-3">
          <div className="rounded-2xl border bg-slate-50 p-5">
            <div className="text-xs font-semibold uppercase tracking-[0.14em] text-slate-500">
              Final Risk Level
            </div>
            <div className="mt-2 text-3xl font-semibold">
              {decision.final_risk_level || "High"}
            </div>
          </div>
          <div className="rounded-2xl border bg-slate-50 p-5">
            <div className="text-xs font-semibold uppercase tracking-[0.14em] text-slate-500">
              Experts Reviewed
            </div>
            <div className="mt-2 text-3xl font-semibold">
              {visibleExperts.length}/3
            </div>
          </div>
          <div className="rounded-2xl border bg-slate-50 p-5">
            <div className="text-xs font-semibold uppercase tracking-[0.14em] text-slate-500">
              Human Review
            </div>
            <div className="mt-2 text-3xl font-semibold">
              {decision.human_review_required ? "Required" : "No"}
            </div>
          </div>
        </div>

        <div className="mt-8 space-y-5">
          <div className="rounded-2xl border bg-white p-6">
            <div className="text-sm font-semibold uppercase tracking-[0.14em] text-slate-500">
              Council Rationale
            </div>
            <p className="mt-3 text-base leading-7 text-slate-700">
              {decision.final_rationale || "The council identified material safety concerns that should be addressed before deployment."}
            </p>
          </div>

          <div className="rounded-2xl border-l-4 border-blue-500 bg-blue-50 p-6">
            <div className="text-sm font-semibold uppercase tracking-[0.14em] text-blue-700">
              Recommended Next Step
            </div>
            <p className="mt-2 text-base text-slate-700">
              {decision.final_decision === "Reject"
                ? "System must be redesigned before resubmission."
                : decision.final_decision === "Escalate"
                ? "Route this system to human governance review before deployment."
                : decision.final_decision === "Revise"
                ? "Apply the required revisions and resubmit for council review."
                : "Deployment may proceed with standard monitoring."}
            </p>
          </div>
        </div>
      </div>

      {/* expert opinions */}
      <div className="rounded-3xl border bg-white p-6">
        <h2 className="text-2xl font-semibold">Expert Opinions</h2>
        <div className="mt-5 grid gap-4 xl:grid-cols-3">
          {visibleExperts.map(([name, expert]) => (
            <details key={name} className="rounded-2xl border bg-slate-50 p-5">
              <summary className="cursor-pointer list-none">
                <div className="flex items-start justify-between gap-4">
                  <div>
                    <div className="text-lg font-semibold">{name}</div>
                    <div className="mt-2 flex flex-wrap gap-2">
                      <span className="rounded-full bg-white px-3 py-1 text-xs">
                        {expert.overall_status}
                      </span>
                      <span className="rounded-full bg-white px-3 py-1 text-xs">
                        Risk: {expert.risk_level}
                      </span>
                      <span className="rounded-full bg-white px-3 py-1 text-xs">
                        Action: {expert.recommended_action}
                      </span>
                    </div>
                  </div>
                  <span className="text-sm text-blue-600">View</span>
                </div>
              </summary>

              <div className="mt-4 space-y-3 text-sm text-slate-700">
                <p><strong>Confidence:</strong> {expert.confidence_level}</p>
                <p><strong>Summary:</strong> {expert.rationale_summary}</p>
                <p><strong>Human Review:</strong> {expert.requires_human_review ? "Yes" : "No"}</p>
              </div>
            </details>
          ))}
        </div>
      </div>
    </div>
  );
}
